{{/* Resolve a partner's logo from assets//partners/.. Sponsors send logos in whatever format they have, so all three are tried. Returns nothing when there is no logo — callers render a text placeholder. @param {string} year Edition, e.g. "2026" @param {string} code Partner code, e.g. "google" @return {resource.Resource|nil} */}} {{- $img := "" -}} {{- $codes := slice $.code (replace $.code "-" "") -}} {{- if in $.code "tetrapack" -}} {{- $codes = $codes | append (replace $.code "tetrapack" "tetrapak") -}} {{- else if in $.code "tetrapak" -}} {{- $codes = $codes | append (replace $.code "tetrapak" "tetrapack") -}} {{- end -}} {{- range $c := $codes -}} {{- if not $img -}} {{- range $ext := slice "png" "jpg" "webp" -}} {{- if not $img -}} {{- $img = resources.Get (printf "%s/partners/%s.%s" $.year $c $ext) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $img -}} {{- range $y := slice "2026" "2025" "2024" -}} {{- if and (not $img) (ne (printf "%v" $y) (printf "%v" $.year)) -}} {{- range $c := $codes -}} {{- if not $img -}} {{- range $ext := slice "png" "jpg" "webp" -}} {{- if not $img -}} {{- $img = resources.Get (printf "%s/partners/%s.%s" $y $c $ext) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $img -}}
